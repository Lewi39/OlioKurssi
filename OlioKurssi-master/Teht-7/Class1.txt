 private readonly int KorkeinKerros = 6;
        private int Kerros {  get; set; }

        int Nappi = Console.ReadLine().Length;

        public int kerros
        {
            get { return kerros; }
            set
            {
                if (Nappi > KorkeinKerros)
                {
                    Nappi = KorkeinKerros;
                }
                kerros = Nappi;
                if (Nappi < 0)
                {
                    Console.WriteLine("Error: Virheellinen Kerros");
                }
            }
            
        }
